# P3-05: FX / Audio Mix Pass

**Phase:** 3 — Commercial-quality art, UI, and game feel
**Profile:** lanfast (mechanical code edits)
**Files:**
- Create: `src/phaser/systems/FXSystem.js`

**Objective:** Give every combat event distinct, readable feedback through particle effects and priority-based audio mixing. Audio must respect user gesture (no autoplay), support mute, and balance tower fire over combat noise. Effects must never obscure the route or enemy identity.

**Acceptance criteria:**
1. Tower fire: Each tower type has a unique muzzle flash (color/size) that plays on fire.
2. Hit feedback: A white/gold flash appears at hit location; slow hits add a cyan ripple.
3. Damage numbers: Floating text (+N) appears briefly at hit location for armor-breaking hits.
4. Death VFX: Enemy species gets a species-specific death effect (raptor=fast burst, trex=large explosion).
5. Slow VFX: A cyan ripple circle expands from hit point.
6. Meteor: Telegraph ring grows from target; on impact, large explosion + scorch mark on ground + brief screen shake.
7. Chrono: Blue distortion ring expands from target when activating rewind.
8. Audio: Tower fire per type matches visual (tranq=thock, drone=high tick, heli=boom, fence=zap, chrono=whoosh).
9. Audio: Mute toggle (press `M` when not targeting meteor) mutes all audio.
10. Audio: Priority mixing — tower fire (priority 1) pauses lower-priority ambient sounds.
11. Effects never cover more than 30% of the route at any point.
12. `node --check` passes for all modified files.

## Step 1: FXSystem (`src/phaser/systems/FXSystem.js`)

Create a singleton-like Phaser System (extends `Phaser.Plugins.SystemPlugin`):

```javascript
import Phaser from 'phaser';

const FX_TYPES = {
  MuzzleFlash: Symbol('MuzzleFlash'),
  HitFlash: Symbol('HitFlash'),
  DamageNumber: Symbol('DamageNumber'),
  DeathExplosion: Symbol('DeathExplosion'),
  SlowRipple: Symbol('SlowRipple'),
  MeteorTelegraph: Symbol('MeteorTelegraph'),
  MeteorImpact: Symbol('MeteorImpact'),
  ScorchMark: Symbol('ScorchMark'),
  ChronoRing: Symbol('ChronoRing'),
};

const TOWER_COLORS = {
  tranq: '#66bb6a',  // green
  fence: '#ffeb3b',  // yellow
  drone: '#42a5f5',  // blue
  heli: '#ef5350',   // red
  chrono: '#ab47bc', // purple
};

export default class FXSystem extends Phaser.Plugins.BasePlugin {
  constructor(scene) {
    super(scene);
    this._active = new Map();
  }

  /** Spawn a muzzle flash at tower position */
  muzzleFlash(x, y, towerType) {
    const color = TOWER_COLORS[towerType] || '#ffffff';
    const size = towerType === 'heli' ? 16 : towerType === 'fence' ? 24 : 10;
    const particle = this.scene.add.circle(x, y, size, parseInt(color.slice(1), 16))
      .setAlpha(0.9)
      .setScrollFactor(0)
      .setDepth(300)
      .setBlendMode(Phaser.BlendModes.ADD);
    this.scene.tweens.add({
      targets: particle,
      scaleX: 1.5,
      scaleY: 1.5,
      alpha: 0,
      duration: 200,
      ease: 'Power2',
      onComplete: () => { particle.destroy(); },
    });
    this._track(particle, 300);
  }

  /** Spawn hit flash at position */
  hitFlash(x, y, status) {
    const color = status === 'slow' ? '#00e5ff' : '#ffffff';
    const radius = status === 'slow' ? 20 : 12;
    const ring = this.scene.add.circle(x, y, radius, parseInt(color.slice(1), 16))
      .setAlpha(0.8)
      .setStrokeStyle(2, parseInt(color.slice(1), 16))
      .setScrollFactor(0)
      .setDepth(290);
    this.scene.tweens.add({
      targets: ring,
      scaleX: 2,
      scaleY: 2,
      alpha: 0,
      duration: status === 'slow' ? 500 : 250,
      ease: 'Power2',
      onComplete: () => { ring.destroy(); },
    });
    this._track(ring, status === 'slow' ? 600 : 300);

    if (status === 'armorBreak') {
      this.damageNumber(x, y - 15, 'ARMOR BROKEN', '#ff9800');
    }
  }

  /** Show floating damage number */
  damageNumber(x, y, text, color) {
    const t = this.scene.add.text(x, y, text, {
      fontSize: `${14 * this.scene._uiScale()}px`,
      color: color || '#ffeb3b',
      stroke: '#000000',
      strokeThickness: 2,
    }).setScrollFactor(0).setOrigin(0.5).setDepth(310);
    this.scene.tweens.add({
      targets: t,
      y: y - 40,
      alpha: 0,
      duration: 800,
      ease: 'Power1',
      onComplete: () => { t.destroy(); },
    });
    this._track(t, 900);
  }

  /** Enemy death explosion */
  deathExplosion(x, y, species) {
    const count = species === 'trex' ? 20 : species === 'pterano' ? 12 : 8;
    const particleSize = species === 'trex' ? 6 : 3;
    const particles = this.scene.add.particles(0, 0, {
      lifespan: 600,
      speed: { min: 30, max: 80 },
      angle: { min: 0, max: 360 },
      scale: { start: particleSize, end: 0 },
      blendMode: 'ADD',
      emitting: false,
    });
    particles.setPosition(x, y);
    particles.explode(count);
    this.scene.time.delayedCall(700, () => particles.destroy());
    this._track(particles, 700);
  }

  /** Slow ripple effect */
  slowRipple(x, y) {
    const ring = this.scene.add.circle(x, y, 5, 0x00e5ff)
      .setAlpha(0.6)
      .setStrokeStyle(2, 0x00e5ff)
      .setScrollFactor(0)
      .setDepth(280);
    this.scene.tweens.add({
      targets: ring,
      scaleX: 4,
      scaleY: 4,
      alpha: 0,
      duration: 600,
      ease: 'Cubic.easeOut',
      onComplete: () => { ring.destroy(); },
    });
    this._track(ring, 700);
  }

  /** Meteor telegraph — growing ring */
  meteorTelegraph(x, y) {
    const ring = this.scene.add.circle(x, y, 10, 0xff5722)
      .setAlpha(0.8)
      .setStrokeStyle(3, 0xff5722)
      .setScrollFactor(0)
      .setDepth(250);
    this.scene.tweens.add({
      targets: ring,
      scaleX: 6,
      scaleY: 6,
      alpha: 0,
      duration: 1500,
      ease: 'Cubic.easeIn',
      onComplete: () => { ring.destroy(); },
    });
    this._track(ring, 1600);

    // Shadow ellipse
    const shadow = this.scene.add.ellipse(x, y, 20, 20, 0x000000)
      .setAlpha(0.4)
      .setScrollFactor(0)
      .setDepth(240);
    this.scene.tweens.add({
      targets: shadow,
      scaleX: 8,
      scaleY: 8,
      alpha: 0.1,
      duration: 1500,
      ease: 'Cubic.easeIn',
      onComplete: () => { shadow.destroy(); },
    });
    this._track(shadow, 1600);
  }

  /** Meteor impact explosion */
  meteorImpact(x, y) {
    // Flash
    const flash = this.scene.add.circle(x, y, 5, 0xffeb3b)
      .setAlpha(1)
      .setScrollFactor(0)
      .setDepth(350)
      .setBlendMode(Phaser.BlendModes.ADD);
    this.scene.tweens.add({
      targets: flash,
      scaleX: 30,
      scaleY: 30,
      alpha: 0,
      duration: 400,
      ease: 'Power2',
      onComplete: () => { flash.destroy(); },
    });
    this._track(flash, 450);

    // Fire particles
    const fire = this.scene.add.particles(0, 0, {
      lifespan: 800,
      speed: { min: 40, max: 120 },
      angle: { min: 0, max: 360 },
      scale: { start: 8, end: 0 },
      blendMode: 'ADD',
      emitting: false,
    });
    fire.setPosition(x, y);
    fire.explode(30);
    this.scene.time.delayedCall(900, () => fire.destroy());
    this._track(fire, 900);

    // Scorch mark
    this.scorchMark(x, y);

    // Screen shake
    this.scene.cameras.main.shake(200, 0.01);
  }

  /** Scorch mark on ground */
  scorchMark(x, y) {
    const mark = this.scene.add.ellipse(x, y, 60, 60, 0x1a1a1a)
      .setAlpha(0.6)
      .setScrollFactor(0)
      .setDepth(200);
    this.scene.time.delayedCall(2000, () => {
      this.scene.tweens.add({
        targets: mark,
        alpha: 0,
        duration: 1500,
        onComplete: () => { mark.destroy(); },
      });
    });
  }

  /** Chrono rewind distortion ring */
  chronoRing(x, y) {
    const ring = this.scene.add.circle(x, y, 5, 0x66e0ff)
      .setAlpha(0.7)
      .setStrokeStyle(3, 0x66e0ff)
      .setScrollFactor(0)
      .setDepth(290);
    this.scene.tweens.add({
      targets: ring,
      scaleX: 10,
      scaleY: 10,
      alpha: 0,
      duration: 1000,
      ease: 'Cubic.easeOut',
      onComplete: () => { ring.destroy(); },
    });
    this._track(ring, 1100);
  }

  /** Screen shake */
  shake(intensity, duration) {
    this.scene.cameras.main.shake(duration, intensity);
  }

  /** Track effect lifecycle for cleanup */
  _track(obj, timeout) {
    this._active.set(obj, this.scene.time.delayedCall(timeout, () => {
      if (obj && !obj.destroyed) obj.destroy();
      this._active.delete(obj);
    }));
  }

  /** Clear all active effects */
  clear() {
    for (const [obj] of this._active) {
      if (obj && !obj.destroyed) obj.destroy();
    }
    this._active.clear();
  }
}

// Register the plugin
export function registerFXSystem() {
  if (!Phaser.Plugins.FXSystem) {
    Phaser.Plugins.FXSystem = FXSystem;
  }
}

export function addFXSystem(scene) {
  if (!scene.plugins.has('FXSystem')) {
    scene.plugins.add('FXSystem');
  }
  return scene.plugins.retrieve('FXSystem');
}
```

## Step 2: AudioSystem (`src/phaser/systems/AudioSystem.js`)

```javascript
import Phaser from 'phaser';

// Priority levels (lower = more important)
const PRIO = {
  UI: 3,
  Ambient: 2,
  TowerFire: 1,
  Hit: 2,
  Kill: 1,
  Meteor: 0,
  Chrono: 0,
  WaveStart: 1,
  Victory: 0,
  Defeat: 0,
};

export default class AudioSystem extends Phaser.Plugins.BasePlugin {
  constructor(scene) {
    super(scene);
    this._context = null;
    this._muted = false;
    this._masterVolume = 1;
    this._channelVolumes = new Map([
      ['tower', 0.7],
      ['combat', 0.6],
      ['fx', 0.5],
      ['ui', 0.8],
    ]);
    this._nextPlayTime = 0;
    this._cooldown = 50; // ms between very similar sounds
  }

  _getCtx() {
    if (!this._context) {
      this._context = new (window.AudioContext || window.webkitAudioContext)();
    }
    if (this._context.state === 'suspended') {
      this._context.resume();
    }
    return this._context;
  }

  /** Play a sound with priority-based mixing */
  play(key, priority = 2, volume = null) {
    if (this._muted) return;
    const now = performance.now();
    if (now - this._nextPlayTime < this._cooldown && priority >= PRIO.TowerFire) {
      return; // throttle
    }
    this._nextPlayTime = now;

    const ctx = this._getCtx();
    const vol = volume ?? this._getChannelVolume(key);
    const gain = ctx.createGain();
    gain.gain.setValueAtTime(vol, ctx.currentTime);
    gain.connect(ctx.destination);
    this._playSound(ctx, gain, key);
  }

  _getChannelVolume(key) {
    if (key.startsWith('tower')) return this._channelVolumes.get('tower');
    if (key.startsWith('combat')) return this._channelVolumes.get('combat');
    if (key.startsWith('fx')) return this._channelVolumes.get('fx');
    return this._channelVolumes.get('ui', 0.8);
  }

  /** Actual sound synthesis per key */
  _playSound(ctx, gain, key) {
    const now = ctx.currentTime;
    switch (key) {
      // Tower fire per type
      case 'tower-tranq': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(180, now);
        osc.frequency.exponentialRampToValueAtTime(100, now + 0.15);
        gain.gain.setValueAtTime(0.7, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.15);
        break;
      }
      case 'tower-drone': {
        const osc = ctx.createOscillator();
        osc.type = 'square';
        osc.frequency.setValueAtTime(800, now);
        osc.frequency.exponentialRampToValueAtTime(400, now + 0.08);
        gain.gain.setValueAtTime(0.4, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.08);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.08);
        break;
      }
      case 'tower-heli': {
        const osc = ctx.createOscillator();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(80, now);
        osc.frequency.exponentialRampToValueAtTime(40, now + 0.25);
        gain.gain.setValueAtTime(0.5, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.25);
        break;
      }
      case 'tower-fence': {
        const buf = ctx.createBuffer(1, ctx.sampleRate * 0.1, ctx.sampleRate);
        const data = buf.getChannelData(0);
        for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (ctx.sampleRate * 0.02));
        const src = ctx.createBufferSource();
        src.buffer = buf;
        gain.gain.setValueAtTime(0.6, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1);
        src.connect(gain);
        src.start(now);
        break;
      }
      case 'tower-chrono': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(1200, now);
        osc.frequency.exponentialRampToValueAtTime(600, now + 0.12);
        gain.gain.setValueAtTime(0.5, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.12);
        break;
      }
      // Combat sounds
      case 'hit': {
        const osc = ctx.createOscillator();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(300, now);
        osc.frequency.exponentialRampToValueAtTime(150, now + 0.06);
        gain.gain.setValueAtTime(0.3, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.06);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.06);
        break;
      }
      case 'slow': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(600, now);
        osc.frequency.linearRampToValueAtTime(400, now + 0.2);
        gain.gain.setValueAtTime(0.3, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.2);
        break;
      }
      case 'kill': {
        const osc = ctx.createOscillator();
        osc.type = 'square';
        osc.frequency.setValueAtTime(500, now);
        osc.frequency.exponentialRampToValueAtTime(100, now + 0.2);
        gain.gain.setValueAtTime(0.4, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.2);
        break;
      }
      case 'armorBreak': {
        const buf = ctx.createBuffer(1, ctx.sampleRate * 0.15, ctx.sampleRate);
        const data = buf.getChannelData(0);
        for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (ctx.sampleRate * 0.05));
        const src = ctx.createBufferSource();
        src.buffer = buf;
        gain.gain.setValueAtTime(0.5, now);
        const filter = ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.value = 1000;
        src.connect(filter);
        filter.connect(gain);
        src.start(now);
        break;
      }
      // Meteor
      case 'meteorTelegraph': {
        const osc = ctx.createOscillator();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(200, now);
        osc.frequency.linearRampToValueAtTime(800, now + 1.5);
        gain.gain.setValueAtTime(0.15, now);
        gain.gain.linearRampToValueAtTime(0.3, now + 1.4);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 1.5);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 1.5);
        break;
      }
      case 'meteorImpact': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(40, now);
        osc.frequency.exponentialRampToValueAtTime(10, now + 0.8);
        gain.gain.setValueAtTime(0.8, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.8);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.8);
        const buf = ctx.createBuffer(1, ctx.sampleRate * 0.5, ctx.sampleRate);
        const data = buf.getChannelData(0);
        for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (ctx.sampleRate * 0.1));
        const noise = ctx.createBufferSource();
        noise.buffer = buf;
        const ng = ctx.createGain();
        ng.gain.setValueAtTime(0.6, now);
        ng.gain.exponentialRampToValueAtTime(0.01, now + 0.5);
        noise.connect(ng);
        ng.connect(ctx.destination);
        noise.start(now);
        break;
      }
      // Chrono
      case 'chronoActivate': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(1000, now);
        osc.frequency.exponentialRampToValueAtTime(200, now + 0.6);
        gain.gain.setValueAtTime(0.5, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.6);
        break;
      }
      // Game events
      case 'waveStart': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, now);
        osc.frequency.linearRampToValueAtTime(600, now + 0.3);
        osc.frequency.linearRampToValueAtTime(900, now + 0.6);
        gain.gain.setValueAtTime(0.4, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.6);
        break;
      }
      case 'victory': {
        const notes = [523, 659, 784, 1047];
        notes.forEach((freq, i) => {
          const osc = ctx.createOscillator();
          osc.type = 'sine';
          osc.frequency.value = freq;
          const g = ctx.createGain();
          g.gain.setValueAtTime(0, now + i * 0.15);
          g.gain.linearRampToValueAtTime(0.3, now + i * 0.15 + 0.05);
          g.gain.exponentialRampToValueAtTime(0.01, now + i * 0.15 + 0.3);
          osc.connect(g);
          g.connect(ctx.destination);
          osc.start(now + i * 0.15);
          osc.stop(now + i * 0.15 + 0.3);
        });
        break;
      }
      case 'defeat': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, now);
        osc.frequency.linearRampToValueAtTime(100, now + 0.8);
        gain.gain.setValueAtTime(0.4, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.8);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.8);
        break;
      }
      // UI
      case 'place': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.value = 880;
        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.08);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.08);
        break;
      }
      case 'sell': {
        const osc = ctx.createOscillator();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(660, now);
        osc.frequency.exponentialRampToValueAtTime(440, now + 0.1);
        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.1);
        break;
      }
      case 'meteorReady': {
        const osc = ctx.createOscillator();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(440, now);
        osc.frequency.exponentialRampToValueAtTime(880, now + 0.15);
        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
        osc.connect(gain);
        osc.start(now);
        osc.stop(now + 0.2);
        break;
      }
      default: {
        // Generic click
        const buf = ctx.createBuffer(1, ctx.sampleRate * 0.05, ctx.sampleRate);
        const data = buf.getChannelData(0);
        for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (ctx.sampleRate * 0.01));
        const src = ctx.createBufferSource();
        src.buffer = buf;
        gain.gain.setValueAtTime(0.1, now);
        src.connect(gain);
        src.start(now);
      }
    }
  }

  /** Mute / unmute all audio */
  setGlobalMute(muted) {
    this._muted = muted;
  }

  /** Get current mute state */
  isMuted() {
    return this._muted;
  }

  /** Set channel volume */
  setChannelVolume(channel, volume) {
    this._channelVolumes.set(channel, Math.max(0, Math.min(1, volume)));
  }

  /** Cleanup */
  destroy() {
    if (this._context) {
      this._context.close();
      this._context = null;
    }
  }
}

export function addAudioSystem(scene) {
```
